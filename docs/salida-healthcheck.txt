# healthcheck.sh — ejecuciones reales
# Entorno: contenedor Ubuntu 24.04.4 LTS, Bash 5.2.21, sin systemd y sin «ss»,
# ejecutado como root el 2026-10-06. Servicios de prueba en marcha:
# PostgreSQL 16, Redis y un servidor web de Python en 127.0.0.1:8080.
# nginx figura en --servicios pero no está instalado ni en marcha en el
# contenedor: el CRÍTICO «servicio caído» es real.
# --excluir-montaje '/opt/*' deja fuera los montajes de /opt, que pertenecen
# a las herramientas del contenedor de pruebas y no al «servidor».
# En la ejecución con --verbose se descarta la salida de error (2>/dev/null)
# porque sus mensajes de depuración enumeran esos montajes internos; los
# detalles de cada comprobación se escriben en la salida estándar y se ven.
# Salida sin retocar (solo se han eliminado los códigos de color).
